Here is a complete, clean implementation of your Java desktop application following the **MVC (Model-View-Controller)** architecture. It runs in the system tray (next to the clock), handles local file detection in the same folder as the JAR, manages automatic creation of default configuration files, implements logging, and provides full interactivity via right-click and left-click tray actions.

---

### 🏛️ MVC Architecture Overview

1. **Model (`AppModel.java`)**: Manages data state, file I/O (`keepit.json`, `keepit-onload.json`, `jars.json`), keep-alive timestamps (`yyyyMMdd'T'HHmmss`), version info, and logging.
2. **View (`AppView.java`)**: Handles the GUI components including the System Tray icon, right-click popup menu, Mission Control window, and external browser integration.
3. **Controller (`AppController.java`)**: Coordinates actions between Model and View, implementing methods like `fnReload()`, `fnDisableEnable()`, `fnConfig()`, `fnMissionControl()`, and `fnExit()`.

### 💻 Complete Java Source Code

You can place all classes in a single package or file structure. Below is the structured implementation using standard Java Swing and AWT (`SystemTray` requires a desktop environment):

```java
import java.awt.*;
import java.awt.event.*;
import java.io.*;
import java.net.URI;
import java.nio.file.*;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.logging.*;

public class KeepItApp {
    public static void main(String[] args) {
        // Ensure AWT toolkit is initialized for SystemTray
        if (!SystemTray.isSupported()) {
            System.err.println("SystemTray is not supported on this system!");
            System.exit(1);
        }
        
        AppModel model = new AppModel();
        AppView view = new AppView();
        AppController controller = new AppController(model, view);
        
        controller.init();
    }
}

// ==========================================
// MODEL
// ==========================================
class AppModel {
    private static final Logger LOGGER = Logger.getLogger(AppModel.class.getName());
    private File baseDir;
    private File keepitJson;
    private File onloadJson;
    private File jarsJson;
    private boolean enabled = true;

    public AppModel() {
        initFilePaths();
        setupLogger();
        ensureFilesExist();
    }

    private void initFilePaths() {
        try {
            // Get directory where the JAR or running class is located
            File jarPath = new File(KeepItApp.class.getProtectionDomain().getCodeSource().getLocation().toURI());
            if (jarPath.isFile()) {
                baseDir = jarPath.getParentFile();
            } else {
                baseDir = new File(".");
            }
            
            keepitJson = new File(baseDir, "keepit.json");
            onloadJson = new File(baseDir, "keepit-onload.json"); // Handling json counterpart
            jarsJson = new File(baseDir, "jars.json");
        } catch (Exception e) {
            baseDir = new File(".");
            keepitJson = new File(baseDir, "keepit.json");
            onloadJson = new File(baseDir, "keepit-onload.json");
            jarsJson = new File(baseDir, "jars.json");
        }
    }

    private void setupLogger() {
        try {
            FileHandler fh = new FileHandler(new File(baseDir, "keepit.log").getAbsolutePath(), true);
            fh.setFormatter(new SimpleFormatter());
            LOGGER.addHandler(fh);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public void log(String message) {
        LOGGER.info(message);
        System.out.println("[LOG] " + message);
    }

    public void ensureFilesExist() {
        try {
            if (!keepitJson.exists()) {
                String defaultKeepit = "{\n  \"program_name\": \"keepit.jar\",\n  \"current_json\": \"keepit.json\",\n  \"version\": \"1.0.0\",\n  \"keep_alive\": \"" + getCurrentTimestamp() + "\",\n  \"status\": \"active\"\n}";
                Files.writeString(keepitJson.toPath(), defaultKeepit);
                log("Created default keepit.json");
            }

            if (!onloadJson.exists()) {
                String defaultOnload = "{\n  \"config_version\": \"1.0\",\n  \"mode\": \"onload_default\"\n}";
                Files.writeString(onloadJson.toPath(), defaultOnload);
                log("Created default keepit-onload.json");
            }

            if (!jarsJson.exists()) {
                String defaultJars = "[\n  \"keepit.jar\"\n]";
                Files.writeString(jarsJson.toPath(), defaultJars);
                log("Created default jars.json");
            }
        } catch (IOException e) {
            log("Error initializing default files: " + e.getMessage());
        }
    }

    public String getCurrentTimestamp() {
        return LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMdd'T'HHmmss"));
    }

    public File getBaseDir() { return baseDir; }
    public File getKeepitJson() { return keepitJson; }
    public File getOnloadJson() { return onloadJson; }
    public boolean isEnabled() { return enabled; }
    public void setEnabled(boolean enabled) { this.enabled = enabled; }
    
    public String getMissionControlData() {
        try {
            String keepitContent = keepitJson.exists() ? Files.readString(keepitJson.toPath()) : "File missing";
            String onloadContent = onloadJson.exists() ? Files.readString(onloadJson.toPath()) : "File missing";
            return "=== KeepIt Mission Control ===\n\n[keepit.json]:\n" + keepitContent + 
                   "\n\n[keepit-onload.json]:\n" + onloadContent + 
                   "\n\nStatus: " + (enabled ? "ACTIVE" : "DISABLED") + 
                   "\nLast Keep-Alive: " + getCurrentTimestamp();
        } catch (IOException e) {
            return "Error reading configuration files: " + e.getMessage();
        }
    }
}

// ==========================================
// VIEW
// ==========================================
class AppView {
    private TrayIcon trayIcon;
    private MenuItem enableDisableItem;

    public void initTray(ActionListener leftClickListener, ActionListener configListener, 
                         ActionListener reloadListener, ActionListener toggleListener, 
                         ActionListener helpListener, ActionListener versionListener, 
                         ActionListener exitListener) {
        if (!SystemTray.isSupported()) return;

        SystemTray tray = SystemTray.getSystemTray();
        PopupMenu popup = new PopupMenu();

        MenuItem configItem = new MenuItem("Config");
        configItem.addActionListener(configListener);
        popup.add(configItem);

        MenuItem reloadItem = new MenuItem("Reload");
        reloadItem.addActionListener(reloadListener);
        popup.add(reloadItem);

        enableDisableItem = new MenuItem("Disable");
        enableDisableItem.addActionListener(toggleListener);
        popup.add(enableDisableItem);

        MenuItem helpItem = new MenuItem("Help");
        helpItem.addActionListener(helpListener);
        popup.add(helpItem);

        MenuItem versionItem = new MenuItem("Version");
        versionItem.addActionListener(versionListener);
        popup.add(versionItem);

        MenuItem exitItem = new MenuItem("Exit");
        exitItem.addActionListener(exitListener);
        popup.add(exitItem);

        // Create a simple default image for the tray icon
        Image image = Toolkit.getDefaultToolkit().createImage(new byte[0]);
        trayIcon = new TrayIcon(image, "KeepIt Monitor", popup);
        trayIcon.setImageAutoSize(true);
        trayIcon.setToolTip("KeepIt Monitor (Active)");

        // Left click action (Mission Control)
        trayIcon.addMouseListener(new MouseAdapter() {
            @Override
            public void mouseClicked(MouseEvent e) {
                if (e.getButton() == MouseEvent.BUTTON1) { // Left click
                    leftClickListener.actionPerformed(new ActionEvent(e, ActionEvent.ACTION_PERFORMED, "leftClick"));
                }
            }
        });

        try {
            tray.add(trayIcon);
        } catch (AWTException e) {
            e.printStackTrace();
        }
    }

    public void updateToggleLabel(boolean enabled) {
        if (enableDisableItem != null) {
            enableDisableItem.setLabel(enabled ? "Disable" : "Enable");
            trayIcon.setToolTip(enabled ? "KeepIt Monitor (Active)" : "KeepIt Monitor (Disabled)");
        }
    }

    public void showMissionControl(String data) {
        Frame frame = new Frame("KeepIt Mission Control");
        frame.setSize(450, 400);
        TextArea textArea = new TextArea(data, 20, 50, TextArea.SCROLLBARS_VERTICAL_ONLY);
        textArea.setEditable(false);
        frame.add(textArea);
        frame.addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent windowEvent) {
                frame.dispose();
            }
        });
        frame.setVisible(true);
    }

    public void showConfigEditor(File file) {
        Frame frame = new Frame("Config Editor - " + file.getName());
        frame.setSize(400, 300);
        TextArea textArea = new TextArea();
        try {
            if (file.exists()) {
                textArea.setText(Files.readString(file.toPath()));
            }
        } catch (IOException e) {
            textArea.setText("Error reading file");
        }
        
        Button saveButton = new Button("Save");
        saveButton.addActionListener(e -> {
            try {
                Files.writeString(file.toPath(), textArea.getText());
                frame.dispose();
            } catch (IOException ex) {
                ex.printStackTrace();
            }
        });

        Panel panel = new Panel();
        panel.add(saveButton);

        frame.add(textArea, BorderLayout.CENTER);
        frame.add(panel, BorderLayout.SOUTH);
        frame.addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent windowEvent) { frame.dispose(); }
        });
        frame.setVisible(true);
    }

    public void openHelp(File baseDir) {
        try {
            File htmlFile = new File(baseDir, "help.html");
            if (!htmlFile.exists()) {
                Files.writeString(htmlFile.toPath(), "<html><body><h1>KeepIt Help</h1><p>KeepIt Monitor helps keep track of your background services and jar tasks.</p></body></html>");
            }
            if (Desktop.isDesktopSupported()) {
                Desktop.getDesktop().browse(htmlFile.toURI());
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

// ==========================================
// CONTROLLER
// ==========================================
class AppController {
    private final AppModel model;
    private final AppView view;

    public AppController(AppModel model, AppView view) {
        this.model = model;
        this.view = view;
    }

    public void init() {
        model.log("KeepIt application started.");
        view.initTray(
            e -> fnMissionControl(),
            e -> fnConfig(),
            e -> fnReload(),
            e -> fnDisableEnable(),
            e -> fnHelp(),
            e -> fnVersion(),
            e -> fnExit()
        );
    }

    public void fnReload() {
        model.log("Configuration reloaded via fnReload().");
        model.ensureFilesExist();
    }

    public void fnDisableEnable() {
        boolean newState = !model.isEnabled();
        model.setEnabled(newState);
        view.updateToggleLabel(newState);
        model.log("Program status changed to: " + (newState ? "Active" : "Disabled"));
    }

    public void fnConfig() {
        model.log("Opening config editor for keepit-onload.json");
        view.showConfigEditor(model.getOnloadJson());
    }

    public void fnMissionControl() {
        model.log("Opening Mission Control.");
        String data = model.getMissionControlData();
        view.showMissionControl(data);
    }

    public void fnHelp() {
        model.log("Opening Help documentation.");
        view.openHelp(model.getBaseDir());
    }

    public void fnVersion() {
        model.log("Version check requested.");
        final Frame f = new Frame("Version");
        Dialog d = new Dialog(f, "KeepIt Version", true);
        d.setLayout(new FlowLayout());
        d.add(new Label("KeepIt Monitor v1.0.0\nKeep-Alive: " + model.getCurrentTimestamp()));
        Button ok = new Button("OK");
        ok.addActionListener(e -> d.dispose());
        d.add(ok);
        d.setSize(250, 150);
        d.setVisible(true);
    }

    public void fnExit() {
        model.log("Exiting KeepIt application.");
        System.exit(0);
    }
}
```

### 🚀 How to Compile & Run

1. Save the code above into a file named `KeepItApp.java`.
2. Compile the code using the terminal:
```bash
javac KeepItApp.java

```


3. Run the application:
```bash
java KeepItApp

```


4. **Behavior on execution**:
* The app sits quietly in your system tray (next to the clock).
* Automatically generates `keepit.json`, `keepit-onload.json`, and `jars.json` in the same directory.
* **Left-click** the tray icon to open **Mission Control**.
* **Right-click** the tray icon to access **Config**, **Reload**, **Enable/Disable**, **Help**, **Version**, and **Exit**.
